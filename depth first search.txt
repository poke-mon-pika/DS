#include <stdio.h>
#include <conio.h>

// Global variables for easier access across functions
int a[20][20], visited[20], n;

void dfs(int v) {
    int i;
    visited[v] = 1; // Mark current node as visited
    printf("%d ", v); // Print the visited node

    for (i = 1; i <= n; i++) {
        // If there is an edge and the adjacent node is not visited
        if (a[v][i] == 1 && !visited[i]) {
            dfs(i); // Recursive call
        }
    }
}

void main() {
    int i, j, start;
    clrscr(); // Clear screen (Turbo C specific)

    printf("Enter number of vertices: ");
    scanf("%d", &n);

    printf("Enter the adjacency matrix (0s and 1s):\n");
    for (i = 1; i <= n; i++) {
        for (j = 1; j <= n; j++) {
            scanf("%d", &a[i][j]);
        }
    }

    // Initialize visited array to 0
    for (i = 1; i <= n; i++) {
        visited[i] = 0;
    }

    printf("Enter the starting vertex: ");
    scanf("%d", &start);

    printf("\nDFS Traversal: ");
    dfs(start);

    getch(); // Wait for key press (Turbo C specific)
}
