//ARRAY IMPLEMENTATION OF STACK
#include<stdio.h>
#include<conio.h>
#include<stdlib.h>

int s[5],top=-1,i,ch,x;

void main()
{
clrscr();
while(1)
{
printf(" \n1.Push \n2.Pop \n3.Display \n4.Exit\n");
scanf("%d",&ch);

switch(ch)
{
case 1:
if(top==4)
printf("Overflow");
else
{
printf("Enter:");
scanf("%d",&x);
s[++top]=x;
}
break;

case 2:
if(top==-1)
printf("Underflow");
else
printf("Deleted=%d",s[top--]);
break;

case 3:
for(i=top;i>=0;i--)
printf("%d ",s[i]);
break;

case 4:
exit(0);
}
}
}
