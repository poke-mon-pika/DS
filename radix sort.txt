#include <stdio.h>
#include <conio.h>

void radix(int a[], int n) {
    int i, exp = 1, max = a[0];
    int out[50], bucket[10];

    /* Find maximum value */
    for (i = 1; i < n; i++) if (a[i] > max) max = a[i];

    /* Process digits */
    while (max / exp > 0) {
        for (i = 0; i < 10; i++) bucket[i] = 0;
        for (i = 0; i < n; i++) bucket[(a[i] / exp) % 10]++;
        for (i = 1; i < 10; i++) bucket[i] += bucket[i - 1];
        
        /* Place elements in sorted order for current digit */
        for (i = n - 1; i >= 0; i--) 
            out[--bucket[(a[i] / exp) % 10]] = a[i];
            
        for (i = 0; i < n; i++) a[i] = out[i];
        exp *= 10;
    }
}

int main() {
    int i, n, a[50];
    clrscr();
    
    printf("Size: ");   scanf("%d", &n);
    printf("Inputs: "); for (i = 0; i < n; i++) scanf("%d", &a[i]);

    radix(a, n);

    printf("Sorted: "); for (i = 0; i < n; i++) printf("%d ", a[i]);
    getch();
    return 0;
}
